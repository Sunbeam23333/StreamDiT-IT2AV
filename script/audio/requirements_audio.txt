# ============================================================================
# Audio Generation 统一环境依赖
#
# 整合以下模型框架的依赖:
#   - TangoFlux          (文生音效, 515M)
#   - ACE-Step v1        (文生BGM, 3.5B)
#   - SongGeneration v2  (歌曲生成, 4B)
#   - CosyVoice3         (语音合成, 0.5B)
#
# 安装:
#   pip install -r requirements_audio.txt --no-deps 的方式不推荐
#   推荐:
#     pip install -r requirements_audio.txt
#     pip install descript-audio-codec --no-deps       # SongGeneration 必需, 必须 --no-deps
#     pip install onnxruntime-gpu==1.20.1 --no-deps   # CosyVoice 必需, 必须 --no-deps (1.18.0 与 numpy 2.x 不兼容)
#
# 建议环境:
#   conda create -n audio_gen python=3.12
#   conda activate audio_gen
#   pip install torch torchaudio torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements_audio.txt
#
# 注意:
#   1. SongGeneration 可能需要额外的 flash-attn 安装:
#      pip install flash-attn --no-build-isolation
#   2. CosyVoice 需要额外的 onnxruntime-gpu:
#      pip install onnxruntime-gpu==1.20.1 --no-deps
#   3. 如果有版本冲突, 可分环境安装:
#      - env_tangoflux: TangoFlux (最轻量, 依赖最少)
#      - env_ace_step:  ACE-Step (需要 diffusers>=0.33.0)
#      - env_songgen:   SongGeneration (需要 diffusers==0.27.2)
#      - env_cosyvoice: CosyVoice (需要 deepspeed, onnxruntime)
# ============================================================================
